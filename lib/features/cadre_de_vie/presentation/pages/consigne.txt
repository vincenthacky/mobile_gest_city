/**
     * List information with filters and pagination
     */
    public function getInformation(Request $request)
    {
        try {
            $query = Report::query();

            if ($request->has('status')) {
                $query->where('status', $request->query('status'));
            }
            if ($request->has('priority')) {
                $query->where('priority', $request->query('priority'));
            }
            if ($request->has('report_type')) {
                $query->where('report_type', $request->query('report_type'));
            }

            $query->where('report_scope', 'information');

            $perPage = (int)$request->query('per_page', 15);
            $infos = $query->with(['images', 'user'])->orderBy('created_at', 'desc')->paginate($perPage);

            return $this->responseSuccessPaginate(InformationResource::collection($infos), 'Information retrieved successfully.');
        } catch (Throwable $e) {
            return $this->responseError($e->getMessage(), 500);
        }
    }
