
/**
     * Synchronisation intelligente basée sur les checksums
     *
     * @param Request $request
     * - is_initializing: bool - true pour la première sync (envoie tout)
     * - global_checksum: string - checksum global du client
     * - items_checksums: array - [{id: uuid, checksum: string}]
     * - filters: object - filtres appliqués (status, search, etc.)
     */
    public function syncProjects(Request $request)
    {
        try {
            $isInitializing = $request->query('is_initializing') === 'true';
            $clientGlobalChecksum = $request->input('global_checksum');
            $clientItemsChecksums = $request->input('items_checksums', []);

            // Récupérer les filtres du client
            $filters = [
                'status' => $request->input('filters.status'),
                'search' => $request->input('filters.search'),
                'created_by' => $request->input('filters.created_by'),
                'sort' => $request->input('filters.sort'),
            ];

            // Construire la requête avec les mêmes filtres que getProjects
            $query = $this->buildProjectQuery($filters);

            // Charger toutes les données (pas de pagination pour la sync)
            $projects = $query->get();

            // Calculer le checksum de chaque projet
            $serverItemsWithChecksums = $projects->map(function ($project) {
                return [
                    'id' => $project->id,
                    'checksum' => model_checksum($project),
                    'data' => ProjectResource::make($project)
                ];
            });

            // Calculer le checksum global côté serveur
            $serverGlobalChecksum = calculate_checksum(
                $serverItemsWithChecksums->pluck('checksum')->toArray()
            );

            // CAS 1: Première synchronisation - Envoyer tout
            if ($isInitializing) {
                return $this->responseSuccess([
                    'sync_type' => 'full',
                    'global_checksum' => $serverGlobalChecksum,
                    // 'items' => $serverItemsWithChecksums->map(fn($item) => [
                    //     'id' => $item['id'],
                    //     'checksum' => $item['checksum'],
                    //     'data' => $item['data']
                    // ]),
                    'items' => $serverItemsWithChecksums->map(fn($item) => $item['data'])->toArray(),
                    'total' => $projects->count(),
                    'timestamp' => now()->toIso8601String()
                ], 'Full synchronization completed');
            }

            // CAS 2: Checksum global identique - Aucune modification
            if ($clientGlobalChecksum === $serverGlobalChecksum) {
                return $this->responseSuccess([
                    'sync_type' => 'none',
                    'global_checksum' => $serverGlobalChecksum,
                    'message' => 'Local database is already synchronized',
                    'timestamp' => now()->toIso8601String()
                ], 'No synchronization needed');
            }

            // CAS 3: Différences détectées - Synchronisation différentielle
            $clientChecksumsMap = collect($clientItemsChecksums)->keyBy('id');
$toAdd = []; // Nouveaux projets
            $toUpdate = []; // Projets modifiés
            $toDelete = []; // Projets supprimés

            // Identifier les ajouts et modifications
            foreach ($serverItemsWithChecksums as $serverItem) {
                $clientItem = $clientChecksumsMap->get($serverItem['id']);

                if (!$clientItem) {
                    // Nouveau projet côté serveur
                    $toAdd[] = [
                        // 'id' => $serverItem['id'],
                        // 'checksum' => $serverItem['checksum'],
                        // 'data' =>
                        $serverItem['data']
                    ];
                } elseif ($clientItem['checksum'] !== $serverItem['checksum']) {
                    // Projet modifié
                    $toUpdate[] = [
                        // 'id' => $serverItem['id'],
                        // 'checksum' => $serverItem['checksum'],
                        // 'data' =>
                        $serverItem['data']
                    ];
                }
            }

            // Identifier les suppressions
            $serverIds = $serverItemsWithChecksums->pluck('id')->toArray();
            foreach ($clientChecksumsMap as $clientId => $clientItem) {
                if (!in_array($clientId, $serverIds)) {
                    $toDelete[] = $clientId;
                }
            }

            return $this->responseSuccess([
                'sync_type' => 'differential',
                'global_checksum' => $serverGlobalChecksum,
                'changes' => [
                    'added' => $toAdd,
                    'updated' => $toUpdate,
                    'deleted' => $toDelete,
                ],
                'stats' => [
                    'added_count' => count($toAdd),
                    'updated_count' => count($toUpdate),
                    'deleted_count' => count($toDelete),
                    'total_server' => $projects->count(),
                ],
                'timestamp' => now()->toIso8601String()
            ], 'Differential synchronization completed');
        } catch (Throwable $e) {
            return $this->responseError($e->getMessage(), 500);
        }
    } 


    private function buildProjectQuery(array $filters)
    {
        $query = Project::query();

        // Appliquer les filtres
        if ($filters['status']) {
            $statusMap = [
                'open' => 'vote_open',
                'not_open' => 'vote_not_open',
                'closed' => 'vote_closed',
                'project_accepted' => 'project_accepted',
                'project_rejected' => 'project_rejected',
            ];
            $query->where('status', $statusMap[$filters['status']] ?? $filters['status']);
        }

        if ($filters['search']) {
            $query->whereRaw('LOWER(title) LIKE ?', ['%' . strtolower($filters['search']) . '%']);
        }

        if ($filters['created_by']) {
            $query->where('created_by', $filters['created_by']);
        }

        if ($filters['sort']) {
            $query->orderByDesc('total_score');
        }

        // Eager loading pour éviter N+1
        $query->with([
            'images',
            'user',
            'votes' => function ($q) {
                $q->select('id', 'project_id', 'user_id', 'vote_type', 'created_at');
            },
            'votes.user:id,full_name,email,phone,image_url'
        ])
            ->withCount([
                'votes as positive_votes_count' => function ($q) {
                    $q->where('vote_type', 'yes');
                },
                'votes as negative_votes_count' => function ($q) {
                    $q->where('vote_type', 'no');
                },
                'votes as neutral_votes_count' => function ($q) {
                    $q->whereNull('vote_type');
                },
                'votes as yes_to_subject_votes_count' => function ($q) {
                    $q->where('vote_type', 'yes_to_subject');
                }
            ]);

        return $query;
    }
